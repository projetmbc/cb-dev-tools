import sys
import io

def simulate_code_session(code_lines):
    """
    Exécute une liste de lignes de code arbitraires ligne par ligne,
    en conservant le contexte et en affichant un rendu de type terminal.
    """
    print("\033[1mPython 3.10.x (interactive session)\033[0m")
    print("Type \"help\", \"copyright\" or \"license\" for more information.\n")

    # Dictionnaire pour stocker l'état (variables, fonctions, imports) d'une ligne à l'autre
    global_context = {}

    for line in code_lines:
        line = line.strip()
        # Ignorer les lignes vides ou les commentaires seuls
        if not line or line.startswith("#"):
            continue

        # 1. Affichage du prompt du terminal
        print(f"\033[36m>>>\033[0m {line}")

        # 2. Redirection de la sortie standard (pour capturer les print() du code)
        old_stdout = sys.stdout
        sys.stdout = captured_output = io.StringIO()

        try:
            # 3. Tentative d'évaluation en tant qu'expression (pour afficher le retour automatiquement)
            # Si c'est une affectation (ex: x = 10) ou une instruction (ex: def, import),
            # eval() lèvera une SyntaxError, on basculera alors sur exec().
            try:
                result = eval(line, global_context)
                if result is not None:
                    print(repr(result))
            except SyntaxError:
                exec(line, global_context)

            # Récupération et restauration de la sortie standard
            sys.stdout = old_stdout
            output_val = captured_output.getvalue()
            if output_val:
                print(output_val, end="")

        except Exception as e:
            # Restauration de sys.stdout en cas d'erreur
            sys.stdout = old_stdout
            # Affichage de l'exception en rouge
            print(f"\033[31m{type(e).__name__}: {e}\033[0m")

        print() # Ligne vide pour aéreracous

from pathlib import Path

from aboutmeta.tool.web import get_text_from

# --------------- #
# -- CONSTANTS -- #
# --------------- #

_URL_TEMPL_SPDX_LICENSE_TEXT = (
    "https://raw.githubusercontent.com/spdx/"
    "license-list-data/main/text/{}.txt"
)



###
# prototype::
#     folder : the path to an existing folder in which to add
#              the path::''LICENSE.txt'' file.
#     erase  : if this option is set to ''True'', it allows to
#              delete an existing final file in order to create
#              a new one.
#
#     :action: create or update a path::''LICENSE.txt'' file in
#              the specified folder with the complete text of
#              the license.
###
    def add_license(
        self,
        folder: Path,
        erase : bool = False
    ) -> None:
# Does the folder exist?
        if not folder.is_dir():
            raise IOError(
                f"the class {type(self).__name__} can't create "
                f"the folder:\n{folder}"
            )

# File for the license.
        license_file = folder /  "LICENSE.txt"

# Can we erase an existing final file?
        if license_file.is_file() and not erase:
            raise IOError(
                f"the class {type(self).__name__} is not allowed "
                f"to erase the LICENSE file:\n{license_file}"
            )

# Everything seems to be in order. Let's proceed with the
# recovery, then add the license text.
        license_text = get_text_from(
            _URL_TEMPL_SPDX_LICENSE_TEXT.format(self.std)
        )

        license_file.touch()
        license_file.write_text(license_text)

~~~python
from aboutmeta import AMData, Path

meta = AMData(Path("/full/path/to/about.yaml"))
meta.build()

meta.add_license(
    license_path = "project.licenses.code",
    erase        = True
)

meta.add_license(
    license_path = "project.licenses.manual",
    dir_relpath  = "readme",
    erase        = True
)
~~~
